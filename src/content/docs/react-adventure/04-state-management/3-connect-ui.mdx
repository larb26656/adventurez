บทนี้คือจุดที่หน้าร้านกับ store เจอกัน ยังใช้ไฟล์ route เดิม ไม่มี state ชื่อหน้า

## การ์ดเรียก addItem

`ProductCard` อ่าน `addItem` ทีละฟังก์ชัน แล้วห่อด้วย `useCallback` กด "เพิ่มลงตะกร้า" แล้ว store เปลี่ยน แถบบนที่เรียก `useCartCount()` จะวาดเลขใหม่ทั้งที่การ์ดไม่ได้ส่งของขึ้นไปหา parent

```tsx
import { useCallback } from 'react'
import type { Product } from '../data/products.ts'
import { useCartStore } from '../stores/cartStore.ts'

interface ProductCardProps {
  product: Product
}

export function ProductCard({ product }: ProductCardProps) {
  const addItem = useCartStore((state) => state.addItem)
  const handleAdd = useCallback(() => {
    addItem(product)
  }, [addItem, product])

  return (
    <article className="flex flex-col rounded-2xl border border-stone-200 bg-white p-4 shadow-sm dark:border-stone-700 dark:bg-stone-900">
      <div className="flex h-36 items-center justify-center rounded-xl bg-amber-50 text-6xl dark:bg-stone-800">
        <span role="img" aria-label={product.name}>
          {product.image}
        </span>
      </div>
      <h2 className="mt-3 text-lg font-semibold">{product.name}</h2>
      <p className="text-sm text-stone-500 dark:text-stone-400">
        {product.temperature === 'cold' ? 'เย็น' : 'ร้อน'}
      </p>
      <p className="mt-1 font-bold text-amber-800 dark:text-amber-300">
        ฿{product.price}
      </p>
      <button
        type="button"
        onClick={handleAdd}
        className="mt-4 w-full rounded-lg bg-amber-800 py-2 font-semibold text-white hover:bg-amber-900"
      >
        เพิ่มลงตะกร้า
      </button>
    </article>
  )
}
```

## ป้ายจำนวนบนแถบ

`useCartCount` รวม quantity วงกลมข้างคำว่า "ตะกร้า" ใน `__root.tsx` แสดงค่านั้นเสมอ รวมถึง 0 เพื่อให้เห็นว่ามันผูกกับ store จริง

```ts
import { useCartStore } from '../stores/cartStore.ts'

export function useCartCount() {
  return useCartStore((state) =>
    state.items.reduce((sum, item) => sum + item.quantity, 0),
  )
}
```

## หน้าตะกร้า

หน้านี้ปรับจำนวนทีละแก้ว ลบแถว และโชว์รวมเป็นบาท ปุ่มสั่งชาไปที่ `/checkout` เมื่อมีของ

ไฟล์เต็มอยู่บทหน้าตะกร้าในพาร์ต 3 จุดที่เรียก store คือ:

```tsx
  const items = useCartStore((state) => state.items)
  const updateQuantity = useCartStore((state) => state.updateQuantity)
  const removeItem = useCartStore((state) => state.removeItem)
  const getTotal = useCartStore((state) => state.getTotal)
  const total = getTotal()
```

## กรองทั้งหมด / เย็น / ร้อน

ปุ่มสามปุ่มอยู่หน้าเมนู ค่าเป็น `all`, `cold`, `hot` ป้ายเป็น ทั้งหมด, เย็น, ร้อน `useTeaFilter` ตัด `products` ที่ loader ส่งมา แล้ว `CatalogPage` วาดเฉพาะ `visibleProducts`

กด "เย็น" แล้วการ์ดร้อนหาย กดเพิ่มชานม ป้ายตะกร้าเพิ่ม ทั้งที่ตัวกรองไม่รีเซ็ต
