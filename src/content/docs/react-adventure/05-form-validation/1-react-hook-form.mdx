ถ้าผูกทุกช่องด้วย `useState` ทั้งฟอร์มจะ render ใหม่ทุกตัวอักษร และต้องเขียนเงื่อนไขตรวจเอง React Hook Form ถือ input แบบไม่ควบคุม แล้วอ่านค่าตอนกดส่ง Zod ตรวจตาม schema

`zodResolver` เชื่อมสองตัวเข้าด้วยกัน `z.infer` สร้าง type `LoginFormData` จาก schema ไม่ต้องประกาศซ้ำ

## เงื่อนไขภาษาไทย

- อีเมลว่าง: กรุณากรอกอีเมล
- อีเมลรูปแบบผิด: รูปแบบอีเมลไม่ถูกต้อง
- รหัสว่าง: กรุณากรอกรหัสผ่าน
- รหัสสั้นกว่า 8 ตัว: รหัสผ่านต้องมีอย่างน้อย 8 ตัวอักษร

ฟอร์มใส่ `noValidate` เพื่อไม่ให้เบราว์เซอร์แย่งโชว์กล่อง error ของตัวเอง เราอยากเห็นข้อความจาก Zod

```ts
import { z } from 'zod'

export const loginSchema = z.object({
  email: z.string().min(1, 'กรุณากรอกอีเมล').email('รูปแบบอีเมลไม่ถูกต้อง'),
  password: z
    .string()
    .min(1, 'กรุณากรอกรหัสผ่าน')
    .min(8, 'รหัสผ่านต้องมีอย่างน้อย 8 ตัวอักษร'),
})

export type LoginFormData = z.infer<typeof loginSchema>
```

## จำแค่อีเมล

ไม่มีรายชื่อผู้ใช้ในเซิร์ฟเวอร์ รหัสที่ผ่าน schema ถือว่าเข้าได้ `login` เก็บ `{ email }` ลง store คีย์ `tea-shop-auth` แล้ว `navigate({ to: '/' })` กลับเมนู

```ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

interface AuthUser {
  email: string
}

interface AuthStore {
  user: AuthUser | null
  login: (email: string) => void
  logout: () => void
}

export const useAuthStore = create<AuthStore>()(
  persist(
    (set) => ({
      user: null,
      login: (email) => set({ user: { email } }),
      logout: () => set({ user: null }),
    }),
    { name: 'tea-shop-auth' },
  ),
)
```

## ฟอร์ม

```tsx
import { zodResolver } from '@hookform/resolvers/zod'
import { useNavigate } from '@tanstack/react-router'
import { useForm } from 'react-hook-form'
import type { LoginFormData } from '../schemas/loginSchema.ts'
import { loginSchema } from '../schemas/loginSchema.ts'
import { useAuthStore } from '../stores/authStore.ts'

const fieldClass =
  'mt-1 w-full rounded-lg border border-stone-300 bg-white px-4 py-2 outline-none focus:border-amber-800 focus:ring-2 focus:ring-amber-200 dark:border-stone-600 dark:bg-stone-900'

export function LoginForm() {
  const navigate = useNavigate()
  const login = useAuthStore((state) => state.login)
  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm<LoginFormData>({
    resolver: zodResolver(loginSchema),
  })

  const onSubmit = async (data: LoginFormData) => {
    await new Promise((resolve) => setTimeout(resolve, 400))
    login(data.email)
    await navigate({ to: '/' })
  }

  return (
    <form
      onSubmit={handleSubmit(onSubmit)}
      className="space-y-4 rounded-2xl border border-stone-200 bg-white p-6 dark:border-stone-700 dark:bg-stone-900"
      noValidate
    >
      <div>
        <label htmlFor="email" className="font-medium">
          อีเมล
        </label>
        <input
          id="email"
          type="email"
          autoComplete="email"
          placeholder="you@email.com"
          className={fieldClass}
          {...register('email')}
        />
        {errors.email && (
          <p className="mt-1 text-sm text-red-700" role="alert">
            {errors.email.message}
          </p>
        )}
      </div>

      <div>
        <label htmlFor="password" className="font-medium">
          รหัสผ่าน
        </label>
        <input
          id="password"
          type="password"
          autoComplete="current-password"
          placeholder="อย่างน้อย 8 ตัว"
          className={fieldClass}
          {...register('password')}
        />
        {errors.password && (
          <p className="mt-1 text-sm text-red-700" role="alert">
            {errors.password.message}
          </p>
        )}
      </div>

      <button
        type="submit"
        disabled={isSubmitting}
        className="w-full rounded-lg bg-amber-800 py-3 font-semibold text-white hover:bg-amber-900 disabled:cursor-not-allowed disabled:bg-amber-800/60"
      >
        {isSubmitting ? 'กำลังเข้าสู่ระบบ...' : 'เข้าสู่ระบบ'}
      </button>
    </form>
  )
}
```

แถบบนอ่าน `user` จาก store เดียวกัน เข้าแล้วปุ่ม "เข้าสู่ระบบ" เปลี่ยนเป็น "ออกจากระบบ"
