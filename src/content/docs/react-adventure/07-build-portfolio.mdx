## 1. Project Structure

### โครงสร้างที่แนะนำ

```
src/
├── components/          # UI components ที่ reusable
│   ├── ui/              # Basic UI (Button, Input, Card)
│   ├── layout/          # Layout components (Header, Footer)
│   └── features/        # Feature-specific components
├── contexts/            # React Context providers
├── hooks/               # Custom hooks (useFetch, useCart, etc.)
├── routes/              # TanStack Router routes
├── stores/              # Zustand stores
├── lib/                 # Utilities, helpers, API client
├── schemas/             # Zod schemas
├── types/               # TypeScript types
└── data/                # Mock data (ถ้ายังไม่มี API)

```

## 2. Data Fetching with Custom Hooks

### ทำไมต้อง Custom Hook?

เมื่อมีหลาย component ที่ต้อง fetch data เดียวกัน → สร้าง hook ไว้ใช้ซ้ำ

### สร้าง useFetch Hook

```tsx
// src/hooks/useFetch.ts
import { useState, useEffect, useCallback } from 'react'

interface UseFetchState<T> {
  data: T | null
  isLoading: boolean
  error: Error | null
}

interface UseFetchReturn<T> extends UseFetchState<T> {
  refetch: () => void
}

export function useFetch<T>(url: string): UseFetchReturn<T> {
  const [state, setState] = useState<UseFetchState<T>>({
    data: null,
    isLoading: true,
    error: null
  })

  const fetchData = useCallback(async () => {
    try {
      setState(prev => ({ ...prev, isLoading: true, error: null }))
      const response = await fetch(url)
      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`)
      }
      const data = await response.json()
      setState({ data, isLoading: false, error: null })
    } catch (err) {
      setState({ data: null, isLoading: false, error: err as Error })
    }
  }, [url])

  useEffect(() => {
    fetchData()
  }, [fetchData])

  return {
    ...state,
    refetch: fetchData
  }
}
```

### ใช้ useFetch

```tsx
function ProductList() {
  const { data: products, isLoading, error, refetch } = useFetch<Product[]>(
    'https://api.example.com/products'
  )

  if (isLoading) return <div>กำลังโหลด...</div>
  if (error) return <div>เกิดข้อผิดพลาด: {error.message}</div>

  return (
    <div>
      {products?.map(product => (
        <ProductCard key={product.id} product={product} />
      ))}
    </div>
  )
}
```

### สร้าง useProduct Hook

สำหรับ product-specific logic:


## 3. ตัวอย่าง: Fake API ก่อน

### ใช้ Mock Service

ถ้ายังไม่มี real API, สร้าง mock service ก่อน:

```tsx
// src/lib/api.ts
import type { Product } from '../features/products/types'

const delay = (ms: number) => new Promise(resolve => setTimeout(resolve, ms))

export async function fetchProducts(): Promise<Product[]> {
  await delay(500) // simulate network delay
  return [
    { id: '1', name: 'Wireless Mouse', price: 599, image: '🖱️', category: 'electronics' },
    { id: '2', name: 'Mechanical Keyboard', price: 2490, image: '⌨️', category: 'electronics' },
    { id: '3', name: 'USB-C Hub', price: 1290, image: '🔌', category: 'accessories' }
  ]
}

export async function fetchProduct(id: string): Promise<Product> {
  await delay(300)
  const products = await fetchProducts()
  const product = products.find(p => p.id === id)
  if (!product) throw new Error('Product not found')
  return product
}
```

### อัพเดต useFetch ให้รองรับ function

```tsx
export function useFetch<T>(
  urlOrFn: string | (() => Promise<T>),
  deps: unknown[] = []
): UseFetchReturn<T> {
  const [state, setState] = useState<UseFetchState<T>>({
    data: null,
    isLoading: true,
    error: null
  })

  const fetchData = useCallback(async () => {
    try {
      setState(prev => ({ ...prev, isLoading: true, error: null }))
      const data = typeof urlOrFn === 'function' ? await urlOrFn() : await fetch(urlOrFn).then(r => r.json())
      setState({ data, isLoading: false, error: null })
    } catch (err) {
      setState({ data: null, isLoading: false, error: err as Error })
    }
  }, [urlOrFn, ...deps])

  useEffect(() => {
    fetchData()
  }, [fetchData])

  return { ...state, refetch: fetchData }
}
```

## 5. Async Await Pattern

### การจัดการ async/await ใน hooks

```tsx
// src/hooks/useAsync.ts
import { useState, useCallback } from 'react'

interface AsyncState<T> {
  data: T | null
  isLoading: boolean
  error: Error | null
}

export function useAsync<T>(asyncFunction: () => Promise<T>) {
  const [state, setState] = useState<AsyncState<T>>({
    data: null,
    isLoading: false,
    error: null
  })

  const execute = useCallback(async () => {
    setState({ data: null, isLoading: true, error: null })
    try {
      const data = await asyncFunction()
      setState({ data, isLoading: false, error: null })
    } catch (err) {
      setState({ data: null, isLoading: false, error: err as Error })
    }
  }, [asyncFunction])

  return { ...state, execute }
}
```

```tsx
import { motion } from 'framer-motion'

import { useCartStore } from '../../cart/stores/cartStore'

interface ProductCardProps {
  product: Product
  index: number
```

## Recap

| Pattern | ใช้ทำอะไร |
|---------|-----------|
| **Feature-based structure** | จัดกลุ่ม code ตาม feature แทนประเภทไฟล์ |
| **Custom hooks** | สกัด data fetching logic ไปใช้ซ้ำ |
| **Loading/Error states** | แสดง feedback ให้ user |
| **Skeleton components** | placeholder ระหว่างโหลด |
