พอสร้าง static UI เสร็จแล้ว เราจะเริ่มเพิ่ม **state และ logic** ใน part ถัดไป แต่รู้จัก hooks พื้นฐานไว้ก่อน:

## useEffect — โหลดข้อมูลตอน component mount

```tsx
import { useEffect, useState } from "react"

function ProductList() {
  const [products, setProducts] = useState([])

  useEffect(() => {
    // ทำงานเมื่อ component โหลด (mount)
    fetch("/api/products")
      .then((res) => res.json())
      .then((data) => setProducts(data))
  }, []) // [] = ทำครั้งเดียวตอน mount

  return <div>{/* แสดง products */}</div>
}
```

## useRef — Focus ไปที่ input

```tsx
import { useRef } from "react"

function SearchInput() {
  const inputRef = useRef(null)

  const handleSearch = () => {
    inputRef.current?.focus() // focus ไปที่ input
  }

  return (
    <>
      <input ref={inputRef} type="text" />
      <button onClick={handleSearch}>ค้นหา</button>
    </>
  )
}
```

## useMemo — คำนวณค่าแคชไว้

```tsx
import { useMemo, useState } from "react"

function ProductFilter() {
  const [search, setSearch] = useState("")
  const products = [{ name: "A" }, { name: "B" }, { name: "AB" }]

  // คำนวณใหม่เฉพาะตอน search หรือ products เปลี่ยน
  const filtered = useMemo(
    () => products.filter((p) => p.name.includes(search)),
    [search, products]
  )

  return <div>{filtered.map(/* ... */)}</div>
}
```
