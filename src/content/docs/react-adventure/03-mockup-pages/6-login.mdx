หน้า `/login` เป็น route ไฟล์ `src/routes/login.tsx` ตัวหน้าเองบางมาก มันวางหัวข้อแล้วเรียก `LoginForm`

ช่องอีเมลกับรหัสผ่าน, ข้อความ error ภาษาไทย, และการจำผู้ใช้ อยู่ในฟอร์มของพาร์ต 5 บทนี้แค่ผูกหน้าเข้ากับ router

ไม่มีหน้าสมัครสมาชิก และไม่มีเซิร์ฟเวอร์ให้ตรวจรหัส ข้อความใต้ฟอร์มบอกไว้ตรงๆ ว่าใส่ค่าที่ผ่าน Zod ก็เข้าได้

```tsx
import { createFileRoute } from '@tanstack/react-router'
import { LoginForm } from '../components/LoginForm.tsx'

export const Route = createFileRoute('/login')({
  component: LoginPage,
})

function LoginPage() {
  return (
    <div className="mx-auto max-w-md">
      <h1 className="mb-6 text-center text-3xl font-bold">เข้าสู่ระบบ</h1>
      <LoginForm />
      <p className="mt-4 text-center text-sm text-stone-500">
        ไม่มีเซิร์ฟเวอร์ ใส่อีเมลและรหัสผ่านที่ผ่านเงื่อนไขได้เลย
      </p>
    </div>
  )
}
```
